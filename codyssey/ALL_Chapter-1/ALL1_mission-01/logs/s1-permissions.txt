# S1-2. 권한 실습 (2026-11-05, 사용자 실행 원본, Ubuntu 26.04.1 LTS / bash / WSL2)

## 실험 1: 파일 권한 (hello.sh) — 644 → 755
pc@DESKTOP-QMSIH7V:~/codyssey$ cd ~/codyssey/practice
pc@DESKTOP-QMSIH7V:~/codyssey/practice$ echo 'echo "hi from script"' > hello.sh
pc@DESKTOP-QMSIH7V:~/codyssey/practice$ ls -l hello.sh
-rw-r--r-- 1 pc pc 22 Nov  5 19:27 hello.sh
pc@DESKTOP-QMSIH7V:~/codyssey/practice$ ./hello.sh
bash: ./hello.sh: Permission denied
pc@DESKTOP-QMSIH7V:~/codyssey/practice$ chmod 755 hello.sh
pc@DESKTOP-QMSIH7V:~/codyssey/practice$ ls -l hello.sh
-rwxr-xr-x 1 pc pc 22 Nov  5 19:27 hello.sh
pc@DESKTOP-QMSIH7V:~/codyssey/practice$ ./hello.sh
hi from script
